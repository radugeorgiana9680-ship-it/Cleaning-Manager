function openTimer(id){ timer.clientId = id; timer.start = null; timer.elapsed = 0; timer.running = false; clearInterval(timerInterval); const client = getClient(id); if(!client) return; document.getElementById("timerClientName").textContent = client.name + " — Cleaning Timer"; document.getElementById("timerMessage").innerHTML = `Rate: ${money(client.rate)}/hour
Press Start when you begin cleaning.`; document.getElementById("timerModal").classList.add("show"); updateTimer(); const oldResult = document.getElementById("timerResult"); if(oldResult) oldResult.remove(); } function startTimer(){ if(!timer.clientId) return; const client = getClient(timer.clientId); if(!client) return; timer.start = Date.now() - (timer.elapsed * 1000); timer.running = true; clearInterval(timerInterval); timerInterval = setInterval(updateTimer, 500); document.getElementById("timerMessage").innerHTML = `Timer is running…
${money(client.rate)}/hour`; const oldResult = document.getElementById("timerResult"); if(oldResult) oldResult.remove(); } function stopTimer(){ if(!timer.running) return; timer.elapsed = Math.floor( (Date.now() - timer.start) / 1000 ); timer.running = false; clearInterval(timerInterval); updateTimer(); const client = getClient(timer.clientId); if(!client) return; // EXACT HOURS WORKED const hours = timer.elapsed / 3600; // CLIENT'S HOURLY RATE const rate = Number(client.rate) || 0; // EXACT EARNINGS const earnings = hours * rate; const h = Math.floor(timer.elapsed / 3600); const m = Math.floor((timer.elapsed % 3600) / 60); const s = timer.elapsed % 60; const oldResult = document.getElementById("timerResult"); if(oldResult) oldResult.remove(); const result = document.createElement("div"); result.id = "timerResult"; result.className = "timer-result"; result.innerHTML = `
✨ Cleaning Completed
${client.name}
Time worked
${String(h).padStart(2,"0")}: ${String(m).padStart(2,"0")}: ${String(s).padStart(2,"0")}
Hourly rate
${money(rate)} / hour
You earned
${money(earnings)}
Calculated from the exact time worked.
`; document .querySelector("#timerModal .modal-box") .appendChild(result); document.getElementById("timerMessage").innerHTML = `Timer stopped ✨
You earned ${money(earnings)}`; // SAVE THE COMPLETED CLEANING JOB if(timer.elapsed > 0){ jobs.push({ id: Date.now(), clientId: client.id, date: today(), start: new Date(timer.start) .toLocaleTimeString("en-GB", { hour:"2-digit", minute:"2-digit" }), end: new Date() .toLocaleTimeString("en-GB", { hour:"2-digit", minute:"2-digit" }), // EXACT HOURS hours: hours, // EXACT EARNINGS amount: earnings, paid: "no" }); saveData(); renderAll(); } } function updateTimer(){ let seconds = timer.elapsed; if(timer.running){ seconds = Math.floor( (Date.now() - timer.start) / 1000 ); } const hours = Math.floor(seconds / 3600); const minutes = Math.floor( (seconds % 3600) / 60 ); const secs = seconds % 60; document.getElementById("timerDisplay").textContent = String(hours).padStart(2,"0") + ":" + String(minutes).padStart(2,"0") + ":" + String(secs).padStart(2,"0"); }
